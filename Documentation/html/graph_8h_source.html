225template <class T>
226bool Graph<T>::addFlowEdge(const T &source, const T &dest, double cap) {
227 auto v1 = findVertex(source);
228 auto v2 = findVertex(dest);
229 if (v1 == nullptr || v2 == nullptr) return false;
230 auto e1 = v1->addEdge(v2, cap);
231 auto e2 = v2->addEdge(v1, 0);
232 e1->setReverse(e2);
233 e2->setReverse(e1);
234 return true;
235}
